# Expected dcg verdict for each command, as `VERDICT|command`.
# setup.sh runs this after installing the config, so a calibration that silently
# stops protecting something fails the install instead of passing quietly.
#
# The ALLOW rows are the daily workflow plus the two rules this kit allowlists.
# The DENY rows are what must never become allowed, including the two gaps
# config.toml closes (`git checkout .` and `git restore .`).

ALLOW|git checkout abc123 -- src/foo.ts
ALLOW|git checkout HEAD~1 -- README.md
ALLOW|npx -y @firecrawl/anydoc convert a.pdf
ALLOW|git push origin main
ALLOW|npm install
ALLOW|wrangler deploy
ALLOW|wrangler d1 migrations apply meudb
ALLOW|git stash pop
ALLOW|rm -rf /tmp/scratch
DENY|git checkout .
DENY|git restore .
DENY|git checkout -- src/foo.ts
DENY|git restore src/foo.ts
DENY|git reset --hard origin/main
DENY|git clean -fd
DENY|git branch -D feature/x
DENY|git stash drop
DENY|git push --force origin main
DENY|rm -rf /home/user/Documents
DENY|echo x > /home/user/.bashrc
DENY|wrangler d1 delete meudb
DENY|wrangler r2 bucket delete meubucket
DENY|docker system prune -af
DENY|psql -c 'DROP DATABASE prod'
DENY|mkfs.ext4 /dev/sda1
